// CONSTRUCTORS

//-> Built in compiler constructor
1. Default CONSTRUCTORS

//-> User defined
2. Copy CONSTRUCTORS
3. Parameterized CONSTRUCTORS
4. Non Parameterized CONSTRUCTORS

class Rectangle {
    private :
        int length;
        int breadth;
    
    public :
// Default or Non Parameterized
        Rectangle(){
            length = 0;
            breadth = 0;
        }

// Parameterized
        Rectangle(int l, int b){
            length = l; or setLength(l);
            breadth = b; or setBreadth(b);
        }
} 
// Copy
        Rectangle(Rectangle &rect){
            length = rect.length;
            breadth = rect.breadth;
        }

//};

Rectangle r1;
Rectangle r2(5,6);
Rectangle r3(r2);
